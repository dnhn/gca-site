import { Aside, Steps, Tabs, TabItem } from '@astrojs/starlight/components';

Clone this project to get started:

```shell
git clone https://github.com/dnhn/gca
```

<Tabs>
  <TabItem label="Repository">
    This will install the hook at the repository level, so that it only applies to the current repository.

    <Steps>
    1. Install the hook:
       ```shell
       cd /path/to/your/repo
       source /path/to/gca/install.sh
       ```
    1. **Commit!**
    </Steps>

    To uninstall, simply delete `.git/hooks/post-commit` from the repository.
  </TabItem>
  <TabItem label="Global">
    This will install the hook globally, so that all new repositories will include it. Existing repositories will not be affected.

    <Aside type="caution" title="Warning">This will alter your global Git configuration. Use with caution.</Aside>

    <Steps>
    1. From this project, run:
       ```shell
       make commit
       ```
    1. Initialise a repository with `git init`, new repositories will now include the hook
    1. **Commit!**
    </Steps>

    To stop including the hook in new repositories, run:

    ```shell
    make reset
    ```

    This only de-registers the hook for newly initialised repositories, you need to delete the hook manually from existing repositories.
  </TabItem>
</Tabs>
